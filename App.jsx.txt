const pts = (a) => a.map((p) => p[0].toFixed(1) + "," + p[1].toFixed(1)).join(" ");
function fill(f, front) {
  const g = { ...f };
  g.k2 = g.k2 || g.k; g.a2 = g.a2 || g.a; g.e2 = g.e2 || g.e; g.w2 = g.w2 || g.w;
  g.m = g.m || [(g.s[0] + g.p[0]) / 2, (g.s[1] + g.p[1]) / 2];
  g.f = g.f || (front ? [g.a[0] - 5, g.a[1]] : [g.a[0] + 6, g.a[1]]);
  g.f2 = g.f2 || (front ? [g.a2[0] + 5, g.a2[1]] : [g.a2[0] + 6, g.a2[1]]);
  return g;
}
function Anim({ attr, a, b, on }) {
  if (!on) return null;
  return <animate attributeName={attr} values={`${a};${b};${a}`} dur="2.8s" repeatCount="indefinite" calcMode="spline" keyTimes="0;0.5;1" keySplines="0.45 0 0.55 1;0.45 0 0.55 1" />;
}
function Poly({ A, B, on, ...r }) {
  const a = pts(A), b = pts(B);
  return <polyline points={a} fill="none" strokeLinecap="round" strokeLinejoin="round" {...r}><Anim attr="points" a={a} b={b} on={on} /></polyline>;
}
function Dot({ A, B, on, r, ...rest }) {
  return <circle cx={A[0]} cy={A[1]} r={r} {...rest}><Anim attr="cx" a={A[0]} b={B[0]} on={on} /><Anim attr="cy" a={A[1]} b={B[1]} on={on} /></circle>;
}

function buildDays(planIn, shuffle) {
  const plan = { ...DEFAULT_PLAN, ...(planIn || {}) };
  const kit = plan.equipment && plan.equipment.length ? plan.equipment : [];
  const sh = shuffle || {};
  const lift = [...new Set(plan.liftDays)].filter((d) => d >= 0 && d <= 6).sort((a, b) => weekOrder(a) - weekOrder(b));
  const canPull = optionsFor("hpull", kit).length + optionsFor("vpull", kit).length;
  const FULLBODY = ["fullB", "fullA", "fullC"];
  const split = canPull >= 2 ? (SPLITS[clamp(lift.length, 1, 6)] || SPLITS[4])
    : FULLBODY.slice(0, clamp(lift.length, 1, 3));
  const cardioId = cardioFor(kit);

  // 5 min warm-up + ~4 min movement block, then about 7 min per strength exercise.
  const cardioMins = plan.cardio === "combined" ? clamp(Math.round(plan.mins * 0.25), 8, 15) : 0;
  const stretchMins = plan.stretch === "combined" ? 6 : 0;
  const nBase = clamp(Math.round((plan.mins - 9 - cardioMins - stretchMins) / 7), 2, 6);

  const days = {};
  const week = []; // everything already used this week, so days don't all look the same
  // Pick for one slot: prefer a move not yet used this week, fall back to reuse when the kit is small.
  const pick = (slot, used, seed) => {
    const [pattern, sec] = slot.split(":");
    let opts = optionsFor(pattern, kit).filter((id) => !used.includes(id));
    if (sec && opts.length > 1) opts = opts.slice(1);
    // Reshuffle should give a real alternative, not a worse workout: stay near the best option.
    if (opts.length) { const best = EX[opts[0]].rank; opts = opts.filter((id) => EX[id].rank >= best - 30); }
    const fresh = opts.filter((id) => !week.includes(id));
    const pool = fresh.length ? fresh : opts;
    return pool.length ? pool[seed % pool.length] : null;
  };
  lift.forEach((dow, i) => {
    const key = split[i % split.length];
    const t = TEMPLATES[key];
    const seed = sh[dow] || 0;
    const n = clamp(key.startsWith("full") && plan.mins >= 45 ? nBase + 1 : nBase, 2, 6);

    const used = [];
    t.slots.slice(0, n).forEach((slot) => {
      const got = pick(slot, used, seed);
      if (got) { used.push(got); week.push(got); }
    });
    // The kit couldn't fill every slot, so top the session back up.
    for (let k = 0; used.length < n && k < BACKFILL.length; k++) {
      const got = pick(BACKFILL[k], used, seed);
      if (got) { used.push(got); week.push(got); }
    }

    const items = used.map(mk);
    for (let k = 1; k + 1 < items.length; k += 2) {
      items[k].pair = items[k + 1].id;
      items[k + 1].pair = items[k].id;
    }
    if (cardioMins) items.push({ id: cardioId, min: cardioMins, note: "finisher, easy pace" });
    if (stretchMins) STRETCH_BLOCK.slice(0, 3).forEach((x) => items.push({ ...x }));
    days[dow] = { title: t.title, sub: t.sub, back: t.back, items };
  });

  const rest = [0, 1, 2, 3, 4, 5, 6].filter((d) => !lift.includes(d)).sort((a, b) => weekOrder(a) - weekOrder(b));
  rest.forEach((dow, i) => {
    const first = i === 0;
    const cardioHere = plan.cardio === "own" && first;
    const stretchHere = plan.stretch === "own" && first;
    if (cardioHere && stretchHere) {
      days[dow] = { title: "Cardio + Mobility", sub: "Steady effort, then loosen up",
        items: [{ id: cardioId, min: clamp(plan.mins - 20, 15, 45) }, ...MOBILITY_DAY.map((x) => ({ ...x }))] };
    } else if (cardioHere) {
      days[dow] = { title: "Cardio", sub: "Steady and easy", items: [{ id: cardioId, min: clamp(plan.mins - 10, 15, 45) }] };
    } else if (stretchHere) {
      days[dow] = { title: "Mobility", sub: "Loosen everything off", items: MOBILITY_DAY.map((x) => ({ ...x })) };
    } else if (plan.walk) {
      days[dow] = { title: "Rest + Walk", sub: "Long walk, no lifting", items: [{ id: "walk", min: 45 }] };
    } else {
      days[dow] = { title: "Rest", sub: "Nothing scheduled. Recovery is part of the plan.", items: [] };
    }
  });
  return days;
}

/* ---------- training logic ---------- */
function lastFor(sessions, id) {
  for (let i = sessions.length - 1; i >= 0; i--) {
    const it = sessions[i].items.find((x) => x.id === id && x.sets && x.sets.some((q) => q.done));
    if (it) return { date: sessions[i].date, sets: it.sets.filter((q) => q.done) };
  }
  return null;
}
// Time off changes what the app puts on the bar.
//  up to 10 days   - normal progression, nudge up when every rep was hit
//  11 to 20 days   - hold exactly where you left off
//  3 to 6 weeks    - come back 10% lighter
//  6 weeks or more - come back 20% lighter
const GAP_HOLD = 10, GAP_DROP = 21, GAP_BIG = 42;
function daysBetween(a, b) { return Math.round((keyDate(b) - keyDate(a)) / 86400000); }
function layoff(days) {
  if (days <= GAP_HOLD) return null;
  if (days < GAP_DROP) return { pct: 1, kind: "hold" };
  if (days < GAP_BIG) return { pct: 0.9, kind: "drop" };
  return { pct: 0.8, kind: "reset" };
}
const spanWords = (d) => {
  if (d < 14) return `${d} days`;
  const w = Math.round(d / 7);
  return w < 9 ? `${w} weeks` : `${Math.round(d / 30)} months`;
};
function suggest(it, last, today) {
  if (!last) return { w: it.kg, note: it.kg == null && !it.bw ? "First time logging this one. Pick a weight you could lift for 2–3 more reps." : null };
  const gap = today ? daysBetween(last.date, today) : 0;
  const off = layoff(gap);
  const w = Math.max(...last.sets.map((q) => Number(q.w) || 0));
  const inc = (EX[it.id] && EX[it.id].inc) || 2.5;
  const allTop = last.sets.length >= it.sets && last.sets.every((q) => q.r >= it.hi);
  const when = `Last done ${fmtShort(last.date)}, ${spanWords(gap)} ago.`;

  if (off) {
    if (it.bw) {
      return { w: 0, note: off.kind === "hold"
        ? `${when} Same reps as last time. No hero sets.`
        : `${when} Cut the reps by about a third today and build back up.` };
    }
    const back = Math.max(0, r25(w * off.pct));
    if (off.kind === "hold") return { w, note: `${when} Same weight as last time, ${w} kg. Earn the next jump.` };
    if (off.kind === "drop") return { w: back, note: `${when} Starting 10% lighter at ${back} kg. You'll be back at ${w} kg inside a fortnight — that's how it's meant to go.` };
    return { w: back, note: `${when} Back to ${back} kg to rebuild the movement first. Strength comes back much faster than it was built.` };
  }

  if (it.bw) return { w: 0, note: allTop ? `Hit ${it.hi} on every set last time. Slow the lowering right down to make it harder.` : null };
  if (allTop) return { w: w + inc, note: `Every rep hit last time at ${w} kg. Up to ${w + inc} kg today.` };
  if (last.sets.some((q) => q.r < it.lo)) return { w, note: `Stay at ${w} kg and own it. Holding strength in a deficit is a win.` };
  return { w, note: `Same ${w} kg, chase one more rep somewhere.` };
}
function prefill(it, sessions, back, today) {
  if (it.min || it.sec) return { ...it, done: false };
  const last = lastFor(sessions, it.id);
  let { w, note: hint } = suggest(it, last, today);
  if (!it.bw && !w && it.kg == null) { w = null; hint = "No weight recorded for this one yet. Pick one and log it."; }
  if (it.tip) hint = hint ? `${it.tip} ${hint}` : it.tip;
  if (back === "tight" && SPINE.includes(it.id) && w) { w = r25(w * 0.9); hint = "Back's tight: 10% lighter today, and stop two reps shy of failure."; }
  return { ...it, n: it.sets, hint, last, sets: Array.from({ length: it.sets }, () => ({ w: w == null ? "" : w, r: it.hi, done: false })) };
}
function bestFor(sessions, id) {
  let b = 0;
  sessions.forEach((s) => s.items.forEach((it) => { if (it.id === id && it.sets) it.sets.forEach((q) => { if (q.done && Number(q.w) > b) b = Number(q.w); }); }));
  return b;
}

  const tHour = trainTime ? Number(trainTime.split(":")[0]) : 9;
  if (hour <= tHour && !trained && !restDay) return trainTime
    ? `${trainTime}, ${name}. ${plan.title} is loaded. Every session is a step to ${target}.`
    : `${plan.title} today, ${name}. Whenever it suits. Every session is a step to ${target}.`;
  if (done >= total - 1) return `Nearly a perfect day. Finish it off. This is how ${target} happens.`;
  if (trained && done < total / 2) return "Session's done. Don't leave the rest on the table: protein, water, and get up from that desk.";
  if (weekCount >= 4) return `${weekCount} sessions this week. That's the guy who gets to ${target}.`;
  if (hour >= 17) return trainTime
    ? `Stretch tonight, then sleep. Tomorrow's ${trainTime} starts right now.`
    : "Stretch tonight, then sleep. Tomorrow starts right now.";
  const pool = [
    goal ? `${toGo} kg to go. Chip away. Today's payoff: ${gain}.` : `Today's payoff: ${gain}. Chip away.`,
    "Training is the signal, protein is the material. Miss the 130 g and the muscle pays for it.",
    "Been sitting an hour? Your back and your heart are paying for it. Two-minute desk reset, now.",
    `Consistency wins. Every tick today buys you ${gain}.`,
  ];
  return pool[(dow + hour) % pool.length];
}

function Stepper({ value, onChange, step = 1, min = 0, unit, width = 64, dec = false }) {
  const v = value === "" || value == null ? "" : value;
  const bump = (d) => { const n = Math.max(min, Math.round(((Number(v) || 0) + d) * 100) / 100); onChange(n); };
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 4 }}>
      <button aria-label="decrease" onClick={() => bump(-step)} style={{ width: 38, height: 44, borderRadius: 10, background: C.deep, display: "grid", placeItems: "center" }}><Minus size={16} /></button>
      <div style={{ position: "relative" }}>
        <input className="num inp" inputMode={dec ? "decimal" : "numeric"} value={v} placeholder="—"
          onChange={(e) => { const t = e.target.value.replace(",", "."); onChange(t === "" ? "" : isNaN(Number(t)) ? v : t); }}
          onBlur={(e) => { if (e.target.value !== "") onChange(Number(e.target.value)); }}
          style={{ width, textAlign: "center", fontSize: 20, padding: "8px 4px", paddingRight: unit ? 18 : 4 }} />
        {unit && <span style={{ position: "absolute", right: 6, top: 14, fontSize: 11, color: C.mist }}>{unit}</span>}

      <button aria-label="increase" onClick={() => bump(step)} style={{ width: 38, height: 44, borderRadius: 10, background: C.deep, display: "grid", placeItems: "center" }}><Plus size={16} /></button>
    </div>
  );
}

  if (summary) {
    return (
      <div className="wrap" style={{ paddingTop: 40 }}>
        <Sunrise size={40} color={C.sun} />
        <h1 className="h1" style={{ marginTop: 14 }}>{summary.line}</h1>
        <div style={{ display: "flex", gap: 28, marginTop: 24 }}>
          <div><div className="num" style={{ fontSize: 34 }}>{summary.sets}</div><div className="sub">sets</div></div>
          <div><div className="num" style={{ fontSize: 34 }}>{summary.vol.toLocaleString()}</div><div className="sub">kg moved</div></div>
          <div><div className="num" style={{ fontSize: 34 }}>{summary.mins}</div><div className="sub">minutes</div></div>
        </div>
        <div style={{ marginTop: 22, paddingLeft: 14, borderLeft: `3px solid ${C.sun}` }}>
          <div className="sub">What that just bought you</div>
          <div style={{ fontSize: 18, marginTop: 4 }}>{summary.gain}</div>
        </div>
        {summary.prs.length > 0 && (
          <div style={{ marginTop: 24, padding: 16, borderRadius: 16, background: C.panel }}>
            <div style={{ fontWeight: 700, color: C.sun }}>New bests</div>
            {summary.prs.map((p) => <div key={p} style={{ marginTop: 4 }}>{p}</div>)}
          </div>
        )}
        <button className="pill ghost" style={{ marginTop: 28 }} onClick={() => setSummary(null)}>Back to training</button>
      </div>
    );
  }

  if (!A) {
    const key = pickDay;
    const plan = sessionFor(key);
    return (
      <div className="wrap" style={{ paddingTop: 28 }}>
        <h1 className="h1">Train</h1>
        <div style={{ display: "flex", gap: 6, overflowX: "auto", margin: "16px 0" }}>
          {[1, 2, 3, 4, 5, 6, 0, "travel"].map((d) => (
            <button key={d} onClick={() => setPickDay(d)} style={{ padding: "8px 12px", borderRadius: 999, flexShrink: 0, background: pickDay === d ? C.sun : C.panel, color: pickDay === d ? C.night : C.paper, fontWeight: 600, fontSize: 14 }}>
              {d === "travel" ? "Hotel" : DOW[d]}
            </button>
          ))}
        </div>
        <div className="h2" style={{ fontSize: 24 }}>{plan.title}</div>
        <div className="sub">{plan.sub}</div>
        {needsWarmup(plan.items) && <div style={{ display: "flex", alignItems: "center", gap: 12, marginTop: 14, padding: "10px 0", borderTop: `1px solid ${C.line}` }}>
          <div style={{ width: 54, height: 54, background: C.panel, borderRadius: 12, padding: 3, flexShrink: 0 }}><Figure pose="hop" /></div>
          <div><div style={{ fontWeight: 600 }}>5 min warm-up + movement block</div><div className="sub">Bike or rower, then {WARMUP.length} moves. Every workout.</div></div>
        </div>}
        <div style={{ marginTop: 16 }}>
          {plan.items.map((it, i) => {
            const last = lastFor(s.sessions, it.id);
            let sug = it.min || it.sec ? null : suggest(it, last, today);
            if (sug && !sug.w && it.kg == null) sug = { w: null };
            return (
              <button key={i} onClick={() => openEx(it.id)} style={{ display: "flex", alignItems: "center", gap: 12, width: "100%", textAlign: "left", padding: "10px 0", borderTop: `1px solid ${C.line}` }}>
                <div style={{ width: 54, height: 54, background: C.panel, borderRadius: 12, padding: 3, flexShrink: 0 }}><Figure pose={EX[it.id].pose} imp={EX[it.id].imp} /></div>
                <div style={{ flex: 1 }}>
                  <div style={{ fontWeight: 600 }}>{EX[it.id].name}</div>
                  <div className="sub">{it.min ? `${it.min} min` : it.sec ? `${it.sec} s` : `${it.sets} × ${it.lo === it.hi ? it.lo : `${it.lo}–${it.hi}`}${it.note ? ` ${it.note}` : ""}`}</div>
                </div>
                {sug && !it.bw && <div className="num" style={{ fontSize: 18, color: sug.w == null ? C.mist : C.paper }}>{sug.w == null ? "set kg" : `${sug.w} kg`}</div>}
              </button>
            );
          })}
        </div>
        {plan.items.length > 0 && (
          <button className="pill go" style={{ width: "100%", marginTop: 20, padding: 16, fontSize: 17 }}
            onClick={() => (plan.back ? setAskBack(key) : start(key))}>Start {plan.title}</button>
        )}
        {plan.items.some((x) => x.sets) && typeof key === "number" && (
          <button className="pill ghost" style={{ width: "100%", marginTop: 10, padding: 13 }}
            onClick={() => setS((o) => ({ ...o, shuffles: { ...(o.shuffles || {}), [key]: ((o.shuffles || {})[key] || 0) + 1 } }))}>
            <Shuffle size={16} style={{ verticalAlign: -3, marginRight: 8 }} />Reshuffle this session
          </button>
        )}
        {plan.items.length === 0 && <p className="sub" style={{ marginTop: 18 }}>Nothing scheduled. Recovery is part of the plan — or change your days under Coach, Your plan.</p>}
        {plan.items[0] && plan.items[0].sets && (
          <button onClick={() => start(key, null, true)} style={{ display: "block", margin: "14px auto 0", color: C.mist, fontSize: 14 }}>Flat today? Do the first lift properly and call it.</button>
        )}
        {askBack !== null && (
          <Sheet onClose={() => setAskBack(null)}>
            <h2 className="h2" style={{ fontSize: 22 }}>Quick back check</h2>
            <p className="sub">Squats and hinges load the lower back. How does it feel right now?</p>
            {[["good", "Good to go", "Full session as planned."], ["tight", "A bit tight", "Spine-loaded lifts drop 10%, stop two reps shy."], ["sore", "Sore", "Swap to a gentle back-care session today."]].map(([v, t, d]) => (
              <button key={v} onClick={() => { start(askBack, v); setAskBack(null); }} style={{ display: "block", width: "100%", textAlign: "left", padding: 16, marginTop: 10, borderRadius: 14, background: C.panel }}>
                <div style={{ fontWeight: 700 }}>{t}</div><div className="sub">{d}</div>
              </button>
            ))}
            <p className="sub" style={{ marginTop: 14 }}>If soreness sticks around more than a week or runs down a leg, get it looked at.</p>
          </Sheet>
        )}
      </div>
    );
  }

  /* active session */
  const Timerbar = timer ? (
    <div style={{ position: "fixed", left: 0, right: 0, bottom: 76, zIndex: 30, display: "flex", justifyContent: "center", padding: "0 18px" }}>
      <div style={{ background: left === 0 ? C.sun : C.panel, color: left === 0 ? C.night : C.paper, borderRadius: 999, padding: "10px 12px 10px 20px", display: "flex", alignItems: "center", gap: 14, maxWidth: 520, width: "100%", boxShadow: "0 8px 30px rgba(0,0,0,.4)" }}>
        <Timer size={18} />
        <div style={{ flex: 1 }}>{left === 0 ? "Go again" : timer.label}</div>
        <div className="num" style={{ fontSize: 22 }}>{Math.floor(left / 60)}:{String(left % 60).padStart(2, "0")}</div>
        <button onClick={() => setTimer(null)} style={{ padding: "6px 12px", borderRadius: 999, background: "rgba(0,0,0,.2)" }}>Skip</button>
      </div>
    </div>
  ) : null;

  if (A.phase === "warm") {
    const wid = A.warmId || "bike", wmin = 5;
    const all = WARMUP.every((id) => A.warm[id]) && A.warm.warmup;
    const tick = (k) => setA((a) => ({ ...a, warm: { ...a.warm, [k]: !a.warm[k] } }));
    return (
      <div className="wrap" style={{ paddingTop: 24 }}>
        <div className="sub">{A.title}</div>
        <h1 className="h1" style={{ marginTop: 4 }}>5 min warm-up</h1>
        <div style={{ display: "flex", alignItems: "center", gap: 12, padding: "10px 0", marginTop: 8 }}>
          <button onClick={() => openEx(wid)} style={{ width: 64, height: 64, background: C.panel, borderRadius: 12, padding: 3, flexShrink: 0 }} aria-label={`${EX[wid].name} demo`}><Figure pose={EX[wid].pose} /></button>
          <div style={{ flex: 1 }}>
            <div style={{ display: "flex", gap: 6 }}>
              {[["bike", "Bike"], ["rower", "Rower"]].map(([k, t]) => (
                <button key={k} onClick={() => setA((a) => ({ ...a, warmId: k }))} aria-pressed={wid === k} style={{ padding: "6px 12px", borderRadius: 999, fontSize: 14, fontWeight: 600, background: wid === k ? C.sun : C.panel, color: wid === k ? C.night : C.paper }}>{t}</button>
              ))}
            </div>
            <div className="sub" style={{ marginTop: 4 }}>Easy pace, just raising the temperature</div>
          </div>
          {!A.warm.warmup && <button onClick={() => setTimer({ until: Date.now() + wmin * 60000, label: "Warm-up" })} style={{ padding: 8, color: C.sun }} aria-label="Start 5 minute timer"><Timer size={20} /></button>}
          <Toggle on={!!A.warm.warmup} onClick={() => tick("warmup")} label="Warm-up done" />
        </div>
        <h2 className="h2" style={{ marginTop: 18 }}>Movement block</h2>
        <p className="sub" style={{ margin: "0 0 6px" }}>About 30 seconds each, in order. Tick them off.</p>
        {WARMUP.map((id, i) => (
          <div key={id} style={{ display: "flex", alignItems: "center", gap: 12, padding: "8px 0", borderTop: `1px solid ${C.line}` }}>
            <div className="num" style={{ width: 16, color: C.mist }}>{i + 1}</div>
            <button onClick={() => openEx(id)} style={{ width: 60, height: 60, background: C.panel, borderRadius: 12, padding: 3, flexShrink: 0 }} aria-label={`${EX[id].name} demo`}><Figure pose={EX[id].pose} /></button>
            <div style={{ flex: 1 }}>
              <div style={{ fontWeight: 600 }}>{EX[id].name}</div>
              <div className="sub">{EX[id].cues[0]}</div>
            </div>
            <Toggle on={!!A.warm[id]} onClick={() => tick(id)} label={`${EX[id].name} done`} />
          </div>
        ))}
        <div style={{ display: "flex", gap: 10, marginTop: 20 }}>
          <button className="pill ghost" onClick={() => setA((a) => ({ ...a, phase: "work" }))}>Skip</button>
          <button className="pill go" style={{ flex: 1, opacity: all ? 1 : 0.85 }} onClick={() => setA((a) => ({ ...a, phase: "work" }))}>{all ? "All done, into the session" : "Start the session"}</button>
        </div>
        {Timerbar}
      </div>
    );
  }

  if (A.phase === "done") {
    const save = () => {
      const items = A.items;
      const sets = items.reduce((n, it) => n + (it.sets ? it.sets.filter((q) => q.done).length : it.done ? 1 : 0), 0);
      const vol = Math.round(items.reduce((n, it) => n + (it.sets ? it.sets.filter((q) => q.done).reduce((m, q) => m + (Number(q.w) || 0) * (Number(q.r) || 0), 0) : 0), 0));
      const prs = [];
      items.forEach((it) => {
        if (!it.sets || it.bw) return;
        const prev = bestFor(s.sessions, it.id);
        const now = Math.max(0, ...it.sets.filter((q) => q.done).map((q) => Number(q.w) || 0));
        if (prev > 0 && now > prev) prs.push(`${EX[it.id].name}: ${now} kg (was ${prev})`);
      });
      const mins = Math.max(1, Math.round((Date.now() - A.startedAt) / 60000));
      const rec = { date: A.date, key: A.key, title: A.title, back: A.back, mins, note, warmup: !!(A.warm && A.warm.warmup), movement: WARMUP.filter((id) => A.warm && A.warm[id]).length, items: items.map((it) => ({ id: it.id, lo: it.lo, hi: it.hi, sets: it.sets, done: it.done, min: it.min, sec: it.sec })) };
      setS((o) => ({ ...o, active: null, sessions: [...o.sessions, rec] }));
      setNote("");
      setSummary({ sets, vol, prs, mins, line: DONE_LINES[s.sessions.length % DONE_LINES.length], gain: GAINS[s.sessions.length % GAINS.length] + "." });
    };
    return (
      <div className="wrap" style={{ paddingTop: 30 }}>
        <h1 className="h1">Finish {A.title}</h1>
        <p className="sub">Anything worth remembering for next time? Energy, the back, a weight that felt easy.</p>
        <textarea className="inp" rows={4} value={note} onChange={(e) => setNote(e.target.value)} placeholder="Optional note" style={{ marginTop: 10 }} />
        <div style={{ display: "flex", gap: 10, marginTop: 16 }}>
          <button className="pill ghost" onClick={() => setA((a) => ({ ...a, phase: "work" }))}>Back</button>
          <button className="pill go" style={{ flex: 1 }} onClick={save}>Save session</button>
        </div>
      </div>
    );
  }

  const it = A.items[A.idx];
  const ex = EX[it.id];
  const n = A.items.length;
  const setItem = (patch) => setA((a) => ({ ...a, items: a.items.map((x, i) => (i === a.idx ? { ...x, ...patch } : x)) }));
  const setSet = (j, patch) => setItem({ sets: it.sets.map((q, k) => (k === j ? { ...q, ...patch } : q)) });
  const timed = it.min || it.sec;
  return (
    <div className="wrap" style={{ paddingTop: 18 }}>
      <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
        <div style={{ flex: 1, display: "flex", gap: 4 }}>
          {A.items.map((x, i) => {
            const d = x.sets ? x.sets.every((q) => q.done) : x.done;
            return <div key={i} style={{ flex: 1, height: 5, borderRadius: 3, background: i === A.idx ? C.sun : d ? C.sage : C.line }} />;
          })}
        </div>
        <div className="sub num" style={{ fontSize: 14 }}>{A.idx + 1} / {n}</div>
      </div>
      <div style={{ background: C.panel, borderRadius: 20, height: 230, marginTop: 14, padding: 8, position: "relative" }}>
        <Figure pose={ex.pose} imp={ex.imp} label={ex.name} />
        {A.back === "tight" && SPINE.includes(it.id) && <div style={{ position: "absolute", left: 12, top: 10, fontSize: 12, background: C.deep, padding: "4px 10px", borderRadius: 999, color: C.sun }}>Back: go 10% lighter</div>}
      </div>
      <h1 className="h1" style={{ marginTop: 16, fontSize: 26 }}>{ex.name}</h1>
      <div className="sub" style={{ fontSize: 15 }}>
        {timed ? (it.min ? `${it.min} minutes` : `${it.sec} seconds`) : `${it.n} × ${it.lo === it.hi ? it.lo : `${it.lo}–${it.hi}`}${it.note ? ` ${it.note}` : ""} · rest ${Math.floor(it.rest / 60)}:${String(it.rest % 60).padStart(2, "0")}`}
        {it.note && timed ? ` · ${it.note}` : ""}
      </div>
      {it.pair && EX[it.pair] && <div style={{ fontSize: 14, color: C.sage, marginTop: 4 }}>Superset with {EX[it.pair].name}: alternate sets, rest after both.</div>}
      <ul style={{ paddingLeft: 18, margin: "10px 0 0", lineHeight: 1.55, fontSize: 15 }}>{ex.cues.map((c) => <li key={c}>{c}</li>)}</ul>
      {it.sets && (it.hint || it.last) && <div style={{ marginTop: 12, padding: "10px 14px", borderLeft: `3px solid ${C.sun}`, background: C.deep, fontSize: 14, lineHeight: 1.45 }}>{it.hint ? it.hint + " " : ""}{it.last ? <span style={{ color: C.mist }}> Last time {fmtShort(it.last.date)}: {it.last.sets.map((q) => (it.bw ? q.r : `${q.w}×${q.r}`)).join(", ")}</span> : null}</div>}

      {timed ? (
        <div style={{ display: "flex", gap: 10, marginTop: 18 }}>
          <button className="pill ghost" onClick={() => setTimer({ until: Date.now() + (it.min ? it.min * 60 : it.sec) * 1000, label: ex.name })}><Timer size={16} style={{ verticalAlign: -3, marginRight: 6 }} />Start timer</button>
          <button className="pill" style={{ flex: 1, background: it.done ? C.sage : C.sun, color: C.night }} onClick={() => setItem({ done: !it.done })}>{it.done ? "Done ✓" : "Mark done"}</button>
        </div>
      ) : (
        <div style={{ marginTop: 16 }}>
          <div style={{ display: "flex", gap: 8, fontSize: 12, color: C.mist, padding: "0 0 4px 28px" }}>
            {!it.bw && <div style={{ width: 146, textAlign: "center" }}>kg</div>}
            <div style={{ width: 146, textAlign: "center" }}>{it.id === "carry" ? "metres" : "reps"}</div>
          </div>
          {it.sets.map((q, j) => (
            <div key={j} style={{ display: "flex", alignItems: "center", gap: 8, padding: "6px 0", opacity: q.done ? 0.7 : 1 }}>
              <div className="num" style={{ width: 20, color: C.mist }}>{j + 1}</div>
              {!it.bw && <Stepper value={q.w} step={EX[it.id].inc || 2.5} dec onChange={(v) => setS((o) => ({ ...o, active: { ...o.active, items: o.active.items.map((x, i) => i === o.active.idx ? { ...x, sets: x.sets.map((qq, k) => (k === j || (k > j && !qq.done) ? { ...qq, w: v } : qq)) } : x) } }))} />}
              <Stepper value={q.r} step={1} onChange={(v) => setSet(j, { r: v })} width={52} />
              <Toggle on={q.done} label={`Set ${j + 1} done`} onClick={() => { setSet(j, { done: !q.done }); if (!q.done && j < it.sets.length - 1) setTimer({ until: Date.now() + it.rest * 1000, label: "Rest" }); }} />
            </div>
          ))}
          <button onClick={() => setItem({ sets: [...it.sets, { ...it.sets[it.sets.length - 1], done: false }] })} style={{ color: C.mist, fontSize: 14, padding: "8px 0" }}>+ Add a set</button>
        </div>
      )}

      <div style={{ display: "flex", gap: 10, marginTop: 20, paddingBottom: timer ? 70 : 0 }}>
        <button className="pill ghost" aria-label="Previous exercise" disabled={A.idx === 0} style={{ opacity: A.idx === 0 ? 0.4 : 1 }} onClick={() => setA((a) => ({ ...a, idx: a.idx - 1 }))}><ChevronLeft size={18} /></button>
        {A.idx < n - 1
          ? <button className="pill go" style={{ flex: 1 }} onClick={() => { setA((a) => ({ ...a, idx: a.idx + 1 })); setTimer(null); }}>Next: {EX[A.items[A.idx + 1].id].name} <ChevronRight size={16} style={{ verticalAlign: -3 }} /></button>
          : <button className="pill go" style={{ flex: 1 }} onClick={() => { setA((a) => ({ ...a, phase: "done" })); setTimer(null); }}>Finish session</button>}
      </div>
      <button onClick={() => { if (sure) { setS((o) => ({ ...o, active: null })); setSure(false); } else setSure(true); }} style={{ color: sure ? C.rose : C.mist, fontSize: 13, marginTop: 18 }}>{sure ? "Tap again to discard. Nothing will be saved." : "Discard session"}</button>
      {Timerbar}
    </div>
  );
}

/* ---------- Progress ---------- */
function Progress({ s, setS, today }) {
  const trackWeight = { ...DEFAULT_PLAN, ...(s.plan || {}) }.trackWeight;
  const W = [...s.weighins].sort((a, b) => (a.date < b.date ? -1 : 1));
  const cur = W.length ? W[W.length - 1].kg : s.profile.startKg;
  const [entry, setEntry] = useState(cur);
  const lost = Math.round((s.profile.startKg - cur) * 10) / 10;
  const prev = W.length > 1 ? W[W.length - 2].kg : null;
  const saveW = () => setS((o) => ({ ...o, weighins: [...o.weighins.filter((x) => x.date !== today), { date: today, kg: Number(entry) }] }));
  const ws = weekStart(today);
  const weekCount = s.sessions.filter((x) => x.date >= ws).length;
  const planDays = [0, 1, 2, 3, 4, 5, 6].filter((d) => DAYS[d] && DAYS[d].items.length).length;
  const last7 = Array.from({ length: 7 }).map((_, i) => s.habits[addDays(today, -i)] || blankHabits);
  const avgP = Math.round(last7.reduce((n, h) => n + (h.protein || 0), 0) / 7);
  const avgW = (last7.reduce((n, h) => n + (h.water || 0), 0) / 7000).toFixed(1);
  const lifts = [["squat", 60], ["bench", 40], ["rdl", 40], ["ohp", 20], ["row", 40], ["dbpress", 7]];
  const [csv, setCsv] = useState(null);
  const exportCsv = () => {
    const rows = [["date", "session", "exercise", "set", "kg", "reps"]];
    s.sessions.forEach((x) => x.items.forEach((it) => { if (it.sets) it.sets.forEach((q, j) => { if (q.done) rows.push([x.date, x.title, EX[it.id].name, j + 1, q.w, q.r]); }); else if (it.done) rows.push([x.date, x.title, EX[it.id].name, "", "", it.min ? `${it.min} min` : `${it.sec} s`]); }));
    const text = rows.map((r) => r.join(",")).join("\n");
    setCsv(text);
    try { navigator.clipboard && navigator.clipboard.writeText(text); } catch (e) {}
  };
  return (
    <div className="wrap" style={{ paddingTop: 28 }}>
      <h1 className="h1">Progress</h1>
      {trackWeight && (<>
      <div style={{ display: "flex", alignItems: "baseline", gap: 10, marginTop: 18 }}>
        <span className="num" style={{ fontSize: 56, lineHeight: 1 }}>{cur}</span><span className="sub">kg now</span>
      </div>
      <div className="sub" style={{ marginTop: 4, fontSize: 15 }}>
        <span style={{ color: C.sage, fontWeight: 700 }}>{lost > 0 ? `${lost} kg down` : "Starting line"}</span> from {s.profile.startKg} kg
        {s.profile.goalKg ? ` · ${Math.round((cur - s.profile.goalKg) * 10) / 10} kg to ${s.profile.goalKg}` : ""}
        {prev != null ? ` · ${Math.round((cur - prev) * 10) / 10 > 0 ? "+" : ""}${Math.round((cur - prev) * 10) / 10} since last weigh-in` : ""}
      </div>
      <div style={{ height: 180, marginTop: 14, marginLeft: -18 }}>
        <ResponsiveContainer width="100%" height="100%">
          <LineChart data={W.map((x) => ({ d: fmtShort(x.date), kg: x.kg }))}>
            <XAxis dataKey="d" stroke={C.mist} fontSize={11} tickLine={false} axisLine={false} />
            <YAxis domain={[Math.min(s.profile.goalKg || 999, cur) - 1, s.profile.startKg + 1]} stroke={C.mist} fontSize={11} tickLine={false} axisLine={false} width={40} />
            <ReferenceLine y={s.profile.startKg} stroke={C.line} strokeDasharray="4 4" label={{ value: `Start ${s.profile.startKg}`, fill: C.mist, fontSize: 11, position: "insideTopRight" }} />
            {s.profile.goalKg && <ReferenceLine y={s.profile.goalKg} stroke={C.sage} strokeDasharray="4 4" label={{ value: `Goal ${s.profile.goalKg}`, fill: C.sage, fontSize: 11, position: "insideBottomRight" }} />}
            <Tooltip contentStyle={{ background: C.deep, border: "none", borderRadius: 10, color: C.paper }} />
            <Line type="monotone" dataKey="kg" stroke={C.sun} strokeWidth={3} dot={{ r: 4, fill: C.sun }} />
          </LineChart>
        </ResponsiveContainer>
      </div>
      <div style={{ display: "flex", alignItems: "center", gap: 10, marginTop: 8 }}>
        <Stepper value={entry} step={0.1} dec onChange={setEntry} unit="kg" width={84} />
        <button className="pill go" onClick={saveW}>Log today's weight</button>
      </div>
      <p className="sub" style={{ marginTop: 8 }}>Weigh in Friday mornings, straight after the toilet, before food. Low-water days make the scale lie.</p>
      </>)}

      <h2 className="h2" style={{ marginTop: 28 }}>This week</h2>
      <div style={{ display: "flex", gap: 28, marginTop: 8 }}>
        <div><div className="num" style={{ fontSize: 30 }}>{weekCount}<span style={{ fontSize: 16, color: C.mist }}>/{planDays}</span></div><div className="sub">sessions</div></div>
        <div><div className="num" style={{ fontSize: 30, color: avgP >= s.profile.proteinTarget ? C.sage : C.paper }}>{avgP}<span style={{ fontSize: 16, color: C.mist }}> g</span></div><div className="sub">protein avg</div></div>

const ICSDAY = ["SU", "MO", "TU", "WE", "TH", "FR", "SA"];

function icsEvent({ uid, day, start, mins, title, desc, alarm }) {
  // No set time: an all-day event on that weekday. VALUE=DATE carries no timezone,
  // so it lands on the right day everywhere.
  const nextOn = (() => {
    let k = todayKey();
    for (let i = 0; i < 8; i++) { if (dowOf(k) === day) return k; k = addDays(k, 1); }
    return todayKey();
  })();
  const stamp = nextOn.replace(/-/g, "");
  if (!start) {
    return [
      "BEGIN:VEVENT", `UID:${uid}@sharpen.local`, `DTSTAMP:${stamp}T000000Z`, "SEQUENCE:1",
      `DTSTART;VALUE=DATE:${stamp}`, `DTEND;VALUE=DATE:${addDays(nextOn, 1).replace(/-/g, "")}`,
      `RRULE:FREQ=WEEKLY;BYDAY=${ICSDAY[day]}`, `SUMMARY:${title}`,
      `DESCRIPTION:${(desc || "").replace(/\n/g, "\\n")}`, "END:VEVENT",
    ];
  }
  const [hh, mm] = start.split(":").map(Number);
  const endM = hh * 60 + mm + mins;
  const pad = (n) => String(n).padStart(2, "0");
  const s = `${pad(hh)}${pad(mm)}00`;
  const e = `${pad(Math.floor(endM / 60) % 24)}${pad(endM % 60)}00`;
  return [
    "BEGIN:VEVENT",
    `UID:${uid}@sharpen.local`,
    `DTSTAMP:${stamp}T000000Z`,
    "SEQUENCE:1",
    `DTSTART;TZID=Pacific/Auckland:${stamp}T${s}`,
    `DTEND;TZID=Pacific/Auckland:${stamp}T${e}`,
    `RRULE:FREQ=WEEKLY;BYDAY=${ICSDAY[day]}`,
    `SUMMARY:${title}`,
    `DESCRIPTION:${(desc || "").replace(/\n/g, "\\n")}`,
    ...(alarm ? ["BEGIN:VALARM", "ACTION:DISPLAY", `DESCRIPTION:${title}`, "TRIGGER:PT0M", "END:VALARM"] : []),
    "END:VEVENT",
  ];
}

function buildIcs({ training, desk, stretch }, plan) {
  const lines = ["BEGIN:VCALENDAR", "VERSION:2.0", "PRODID:-//Sharpen//EN", "CALSCALE:GREGORIAN", ...ICS_TZ];
  if (training) {
    const lifts = [...new Set(plan.liftDays)];
    for (let d = 0; d <= 6; d++) {
      const day = DAYS[d];
      if (!day || !day.items.length) continue;
      const isLift = lifts.includes(d);
      lines.push(
        ...icsEvent({
          uid: `sharpen-train-${d}`,
          day: d,
          start: isLift ? (plan.trainTime || "") : (plan.trainTime ? "07:00" : ""),
          mins: isLift ? plan.mins : 45,
          title: `Sharpen: ${day.title}`,
          desc: day.sub + (needsWarmup(day.items) ? "\n\n5 min warm-up, then the movement block, then the session." : ""),
          alarm: true,
        })
      );
    }
  }
  if (desk) {
    [["09:30", "a"], ["11:30", "b"], ["14:30", "c"]].forEach(([t, tag]) => {
      for (let d = 1; d <= 5; d++) {
        lines.push(
          ...icsEvent({
            uid: `sharpen-desk-${d}-${tag}`,
            day: d,
            start: t,
            mins: 5,
            title: "Desk reset (2 min)",
            desc: "Chin tucks, hip flexor stretch, chair back extension, calf raises, marches, core squeeze.",
            alarm: true,
          })
        );
      }
    });
  }
  if (stretch) {
    for (let d = 0; d <= 6; d++) {
      lines.push(
        ...icsEvent({
          uid: `sharpen-stretch-${d}`,
          day: d,
          start: "19:45",
          mins: 15,
          title: "Evening stretch",
          desc: "Hip flexors, hamstrings, figure-4, chest opener, twists, calves, child's pose, core squeeze. Then bed by 8.",
          alarm: true,
        })
      );
    }
  }
  lines.push("END:VCALENDAR");
  return lines.join("\r\n");
}

function download(name, text, type) {
  try {
    const blob = new Blob([text], { type: type || "text/plain;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = name;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 2000);
    return true;
  } catch (e) {
    return false;
  }
}

async function callClaude(apiKey, model, body) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": apiKey,
      "anthropic-version": "2023-06-01",
      "anthropic-dangerous-direct-browser-access": "true",
    },
    body: JSON.stringify({ model: model || "claude-sonnet-5", max_tokens: 800, ...body }),
  });
  if (!res.ok) {
    const t = await res.text().catch(() => "");
    throw new Error(res.status === 401 ? "key" : t.slice(0, 120) || String(res.status));
  }
  const data = await res.json();
  return (data.content || []).filter((b) => b.type === "text").map((b) => b.text).join("\n");
}

function planSummary(s) {
  const plan = { ...DEFAULT_PLAN, ...(s.plan || {}) };
  const n = [...new Set(plan.liftDays)].length;
  const kit = (plan.equipment || []).length;
  const bits = [`${n} lifting ${n === 1 ? "day" : "days"} a week`, `${plan.mins} minutes a session`];
  bits.push(kit ? `${kit} pieces of kit` : "bodyweight only");
  bits.push(plan.cardio === "off" ? "no cardio" : plan.cardio === "own" ? "cardio on its own day" : "cardio in each session");
  bits.push(plan.stretch === "off" ? "no stretching" : plan.stretch === "own" ? "stretching on its own day" : "stretching at the end of each session");
  if (!plan.trackWeight) bits.push("weight not tracked");
  return bits.join(" \u00b7 ") + ".";
}

    missedYesterday: dowOf(yest) !== 0 && !s.sessions.some((x) => x.date === yest) && s.sessions.length > 0,
  });
  const goTrain = (key) => { setPickDay(key); setTab("train"); };
  useEffect(() => { window.scrollTo(0, 0); }, [tab, s.active ? s.active.idx : -1, s.active ? s.active.phase : ""]);

  if (!loaded) return <div className="sh" style={{ display: "grid", placeItems: "center" }}><style>{css}</style><div className="sub">Loading your log…</div></div>;

  if (!s.setupDone) {
    return (
      <div className="sh" style={{ paddingBottom: 0 }}>
        <style>{css}</style>
        <Setup onDone={({ plan, profile, weighins }) => setS((o) => ({
          ...o, setupDone: true, plan, weighins,
          sessions: [], habits: {}, chat: [], calendar: null, shuffles: {},
          profile: { ...o.profile, ...profile },
          whys: [],
        }))} />
      </div>
    );
  }

  const tabs = [["today", "Today", Sunrise], ["train", "Train", Dumbbell], ["progress", "Progress", TrendingUp], ["library", "Library", BookOpen], ["coach", "Coach", MessageCircle]];
  return (
    <div className="sh">
      <style>{css}</style>
      {saveErr && <div style={{ background: C.rose, color: C.night, padding: "8px 18px", fontSize: 14 }}>Couldn't save to this phone's storage. Your changes are on screen but won't survive a reload. Private browsing mode is the usual cause.</div>}
      {tab === "today" && <Today s={s} setS={setS} today={today} goTrain={goTrain} openEx={setExOpen} openRoutine={setRoutine} openTight={() => setTightOpen(true)} habitsDone={done} habitsTotal={checks.length} coach={coach} />}
      {tab === "train" && <Train s={s} setS={setS} today={today} pickDay={pickDay} setPickDay={setPickDay} openEx={setExOpen} />}
      {tab === "progress" && <Progress s={s} setS={setS} today={today} />}
      {tab === "library" && <Library openEx={setExOpen} openTight={() => setTightOpen(true)} />}
      {tab === "coach" && <Coach s={s} setS={setS} today={today} openPlan={() => setPlanOpen(true)} />}
      {exOpen && <ExerciseSheet id={exOpen} onClose={() => setExOpen(null)} />}
      {routine && <RoutineSheet which={routine} onClose={() => setRoutine(null)} />}
      {planOpen && <PlanSheet s={s} setS={setS} onClose={() => setPlanOpen(false)} />}
      {tightOpen && <TightSheet onClose={() => setTightOpen(false)} openEx={(id) => { setTightOpen(false); setExOpen(id); }} />}
      <nav style={{ position: "fixed", left: 0, right: 0, bottom: 0, background: C.deep, borderTop: `1px solid ${C.line}`, zIndex: 40, paddingBottom: "env(safe-area-inset-bottom)" }}>
        <div style={{ display: "flex", maxWidth: 560, margin: "0 auto" }}>
          {tabs.map(([k, t, Icon]) => (
            <button key={k} onClick={() => setTab(k)} aria-current={tab === k ? "page" : undefined} style={{ flex: 1, padding: "10px 0 12px", color: tab === k ? C.sun : C.mist, display: "flex", flexDirection: "column", alignItems: "center", gap: 3, fontSize: 11, fontWeight: 600 }}>
              <Icon size={22} />{t}{k === "train" && s.active ? " ·" : ""}
            </button>
          ))}
        </div>
      </nav>
    </div>
  );
}
